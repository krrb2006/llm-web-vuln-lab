#!/usr/bin/env python3
# S013 本地反射型 XSS 验证辅助脚本
# 仅生成固定目标 http://127.0.0.1:4280 的普通/测试 URL；
# 不自动发起请求、不跟随重定向、不外发数据。实际 JavaScript 执行需人工在浏览器确认。

from urllib.parse import urlencode, urlparse

BASE = "http://127.0.0.1:4280"
PATH = "/"  # 若实际部署路径不同，只修改此路径；主机和端口保持固定

def build_url(q: str) -> str:
    return f"{BASE}{PATH}?{urlencode({'q': q})}"

normal_q = "hello"
test_q = "<script>alert('XSS-S013')</script>"

normal_url = build_url(normal_q)
test_url = build_url(test_q)

# 固定目标校验：只允许 http://127.0.0.1:4280
for u in (normal_url, test_url):
    p = urlparse(u)
    assert p.scheme == "http" and p.hostname == "127.0.0.1" and p.port == 4280, u

print("普通对照 URL：")
print(normal_url)
print()
print("测试 URL：")
print(test_url)
