#!/usr/bin/env python3
import re
import sys
import urllib.error
import urllib.parse
import urllib.request
from html.parser import HTMLParser

BASE_URL = "http://127.0.0.1:4280/lab_cases/S001/index.php"
ALLOWED_HOST = "127.0.0.1"
ALLOWED_PORT = 4280
ALLOWED_PATH = "/lab_cases/S001/index.php"


class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


class ParagraphParser(HTMLParser):
    def __init__(self):
        super().__init__()
        self.in_p = False
        self.buf = []
        self.paragraphs = []

    def handle_starttag(self, tag, attrs):
        if tag.lower() == "p":
            self.in_p = True
            self.buf = []

    def handle_data(self, data):
        if self.in_p:
            self.buf.append(data)

    def handle_endtag(self, tag):
        if tag.lower() == "p" and self.in_p:
            self.paragraphs.append("".join(self.buf))
            self.in_p = False
            self.buf = []


def build_allowed_url(name):
    query = urllib.parse.urlencode({"name": name})
    url = BASE_URL + "?" + query
    parts = urllib.parse.urlsplit(url)
    if parts.scheme != "http":
        raise ValueError("仅允许 http")
    if parts.hostname != ALLOWED_HOST:
        raise ValueError("仅允许主机 127.0.0.1")
    if parts.port != ALLOWED_PORT:
        raise ValueError("仅允许端口 4280")
    if parts.path != ALLOWED_PATH:
        raise ValueError("仅允许固定路径")
    return url


def fetch_html(name):
    url = build_allowed_url(name)
    opener = urllib.request.build_opener(NoRedirect)
    request = urllib.request.Request(
        url,
        method="GET",
        headers={"User-Agent": "S001-audit-poc"},
    )
    with opener.open(request, timeout=5) as response:
        return response.read().decode("utf-8", "replace")


def parse_records(html):
    parser = ParagraphParser()
    parser.feed(html)
    records = set()
    for text in parser.paragraphs:
        match = re.fullmatch(r"ID\s+(\d+)：(.+)", text.strip(), re.DOTALL)
        if match:
            records.add((match.group(1), match.group(2)))
    return records


def main():
    normal_html = fetch_html("Alice")
    test_html = fetch_html("' OR '1'='1")

    normal_records = parse_records(normal_html)
    test_records = parse_records(test_html)

    print("普通请求解析记录:", sorted(normal_records))
    print("测试请求解析记录:", sorted(test_records))

    expected_normal = {("1", "Alice")}
    expected_test = {("1", "Alice"), ("2", "Bob"), ("3", "Carol")}

    if normal_records != expected_normal:
        print("普通业务对照失败：预期仅 (1, Alice)。", file=sys.stderr)
        print("请检查 MariaDB、common/db_connect.php 的 $db、lab_people 预置数据和入口路径。", file=sys.stderr)
        return 2

    if test_records != expected_test:
        print("测试断言失败：预期返回 (1, Alice)、(2, Bob)、(3, Carol)。", file=sys.stderr)
        print("请检查当前代码、HTML 输出格式、数据库错误日志和运行前提。", file=sys.stderr)
        return 3

    print("断言通过：普通请求仅 Alice；测试请求返回全部三行，且该 SELECT 测试不修改数据库。")
    return 0


if __name__ == "__main__":
    try:
        sys.exit(main())
    except urllib.error.HTTPError as exc:
        print("HTTP 错误或被禁止的重定向：", exc, file=sys.stderr)
        sys.exit(4)
    except Exception as exc:
        print("验证无法完成：", exc, file=sys.stderr)
        sys.exit(5)
