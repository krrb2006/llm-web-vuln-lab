```python
import re
import urllib.parse
import urllib.request

BASE = "http://127.0.0.1:4280/lab_cases/S005/index.php"
ROW_RE = re.compile(r'<p>ID\s+(\d+)：([^<]*)</p>')


def fetch(q):
    url = BASE + "?" + urllib.parse.urlencode({"q": q})
    with urllib.request.urlopen(url, timeout=5) as resp:
        body = resp.read().decode("utf-8", "replace")
    rows = [(int(i), name) for i, name in ROW_RE.findall(body)]
    return url, rows


def main():
    normal_q = "Alice"
    test_q = "' OR 1=1 -- "

    normal_url, normal_rows = fetch(normal_q)
    test_url, test_rows = fetch(test_q)

    print("normal url:", normal_url)
    print("normal rows:", normal_rows)
    print("test url:", test_url)
    print("test rows:", test_rows)

    normal_ok = normal_rows == [(1, "Alice")]
    test_names = {name for _, name in test_rows}
    test_ok = {"Alice", "Bob", "Carol"}.issubset(test_names)

    print("normal_expected_only_Alice:", normal_ok)
    print("test_expected_three_preset_people:", test_ok)

    if not (normal_ok and test_ok):
        print("检查：入口可达、common/db_connect.php 的 $db、lab_people 表与 1 Alice/2 Bob/3 Carol 数据。")
        raise SystemExit(1)


if __name__ == "__main__":
    main()
```
