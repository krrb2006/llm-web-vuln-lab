import sys
from urllib.parse import urlencode, urlparse

TARGET = "http://127.0.0.1:4280/"

def validate_target(target: str) -> str:
    parsed = urlparse(target)
    if parsed.scheme != "http":
        raise ValueError("仅允许 http")
    if parsed.hostname != "127.0.0.1":
        raise ValueError("仅允许 127.0.0.1")
    if parsed.port != 4280:
        raise ValueError("仅允许端口 4280")
    if parsed.path not in ("", "/"):
        raise ValueError("仅允许根路径")
    if parsed.query or parsed.fragment:
        raise ValueError("目标不应包含查询串或片段")
    return target

def build_url(target: str, q: str) -> str:
    validate_target(target)
    return target.rstrip("/") + "/?" + urlencode({"q": q})

def main() -> None:
    target = validate_target(TARGET)

    normal_q = "S013-normal-search"
    test_q = "<script>alert('XSS-S013')</script>"

    normal_url = build_url(target, normal_q)
    test_url = build_url(target, test_q)

    print("普通对照URL：")
    print(normal_url)
    print()
    print("测试URL：")
    print(test_url)
    print()
    print("说明：请在浏览器中人工打开上述URL。")
    print("普通对照应只显示搜索词文本；测试URL需人工确认是否出现 alert 弹窗 XSS-S013。")
    print("HTTP响应体出现payload不等同于JavaScript执行；脚本不自动请求、不跟随重定向、不外发数据。")

if __name__ == "__main__":
    try:
        main()
    except Exception as exc:
        print(f"错误：{exc}", file=sys.stderr)
