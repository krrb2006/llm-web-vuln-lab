import urllib.parse
import re
import sys

HOST = "127.0.0.1"
PORT = 4280
PATH = "/lab_cases/S001/index.php"
RECORD_RE = re.compile(r"<p>ID\s*(\d+)：([^<]*)</p>")


def fetch(name):
    conn = http.client.HTTPConnection(HOST, PORT, timeout=5)
    query = urllib.parse.urlencode({"name": name})
    conn.request(
        "GET",
        PATH + "?" + query,
        headers={"Host": "127.0.0.1:4280"},
    )
    resp = conn.getresponse()
    status = resp.status
    body = resp.read().decode("utf-8", "replace")
    conn.close()
    return status, body


def parse_records(body):
    return [(m.group(1), m.group(2)) for m in RECORD_RE.finditer(body)]
