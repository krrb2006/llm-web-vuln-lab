```python
#!/usr/bin/env python3
import argparse
import http.client
import urllib.parse

TARGET_HOST = "127.0.0.1"
TARGET_PORT = 4280
TARGET_BASE = f"http://{TARGET_HOST}:{TARGET_PORT}"

def validate_path(path: str) -> str:
    if not path.startswith("/"):
        raise SystemExit("--path 必须以 / 开头，例如 --path /sample.php")
    if "://" in path or path.startswith("//"):
        raise SystemExit("--path 只能是同一目标主机下的路径，不能是绝对 URL")
    if "?" in path or "#" in path:
        raise SystemExit("--path 不要包含 ? 或 #；查询参数由脚本生成")
    return path

def fetch(path: str, q: str):
    query = urllib.parse.urlencode({"q": q})
    request_path = f"{path}?{query}"
    conn = http.client.HTTPConnection(TARGET_HOST, TARGET_PORT, timeout=10)
    conn.request(
        "GET",
        request_path,
        headers={
            "Host": f"{TARGET_HOST}:{TARGET_PORT}",
            "User-Agent": "S013-local-xss-check",
        },
    )
    resp = conn.getresponse()
    body = resp.read()
    status = resp.status
    location = resp.getheader("Location")
    conn.close()
    return request_path, status, location, body

def main():
    parser = argparse.ArgumentParser(description="S013 本地反射型 XSS 模型验证脚本")
    parser.add_argument("--path", required=True, help="该 PHP 脚本在 127.0.0.1:4280 上的路径，例如 /sample.php")
    args = parser.parse_args()

    path = validate_path(args.path)
    normal_q = "普通搜索词"
    test_q = "<script>alert('S013')</script>"

    print(f"固定目标: {TARGET_BASE}{path}")
    print("说明: 本脚本不自动跟随重定向，不访问外部主机，不执行浏览器。")

    normal_request_path, normal_status, normal_location, normal_body = fetch(path, normal_q)
    normal_url = f"{TARGET_BASE}{normal_request_path}"
    print("\n[普通对照]")
    print(f"URL: {normal_url}")
    print(f"HTTP 状态: {normal_status}")
    if normal_location:
        print(f"Location(未跟随): {normal_location}")
    print("浏览器观察: 应仅显示搜索词文本，不应对照输入产生脚本执行。")

    test_request_path, test_status, test_location, test_body = fetch(path, test_q)
    test_url = f"{TARGET_BASE}{test_request_path}"
    contains_payload = test_q.encode("utf-8") in test_body
    print("\n[测试输入]")
    print(f"URL: {test_url}")
    print(f"HTTP 状态: {test_status}")
    if test_location:
        print(f"Location(未跟随): {test_location}")
    print(f"诊断: 响应体包含原始 payload = {contains_payload}（此结果不等同于 JavaScript 执行）")
    print("人工确认步骤: 用浏览器打开上面的测试 URL，观察是否弹出 S013 提示框。")
    print("成功断言: 仅当人工在浏览器中看到 S013 弹窗/提示框，才可认为该路径实际触发 XSS。")
    print("若未弹窗，需检查 --path 是否正确、服务端是否额外转义、是否有 CSP/重定向或浏览器拦截。")

if __name__ == "__main__":
    main()
```
