def fetch_records(name):

    path = f"{PATH}?name={query}"
    conn = http.client.HTTPConnection(HOST, PORT, timeout=TIMEOUT)
    try:
        conn.request("GET", path, headers={"User-Agent": "S007-local-audit"})
        resp = conn.getresponse()
        status = resp.status
        body = resp.read().decode("utf-8", errors="replace")
    finally:
        conn.close()

    if status != 200:
        raise RuntimeError(f"HTTP status {status}; 不跟随重定向，无法继续判断")

    records = [
        (html.unescape(m.group(1)), html.unescape(m.group(2)).strip())
        for m in pattern.finditer(body)
    ]
    return records, body
